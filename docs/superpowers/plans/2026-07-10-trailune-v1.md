# Trailune v1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a working local-only v1 of Trailune — a SvelteKit travel tracker with trip journals (photos + notes), a reusable packing checklist, per-trip Do's & Don'ts, and a world map of visited countries.

**Architecture:** SvelteKit app with zero backend. All data persists client-side: structured records (trips, journal entries, packing items, do/don't items) in `localStorage` via small repository modules, photo binaries in `IndexedDB` via the `idb` package. A `storage.ts` facade composes these repositories and owns cross-cutting rules (packing-template copy on trip creation, cascade delete). UI is plain SvelteKit routes/components using Svelte 5 runes, styled with a warm passport-themed CSS variable palette. The world map is a static bundled SVG (country paths keyed by ISO code) manipulated directly via the DOM — no map/geo service.

**Tech Stack:** SvelteKit, TypeScript, Svelte 5 (runes), Vitest + jsdom + fake-indexeddb (storage-layer tests), `idb` (IndexedDB wrapper). No backend, no external APIs.

## Global Constraints

- No backend, no server/database, no external map or geo API — all persistence is `localStorage` + `IndexedDB` in the browser (per spec Tech Stack / Non-goals).
- App name is **Trailune**; slogan is **"Track every journey."** — used verbatim in the site header (per spec Naming).
- Visual style is the warm passport aesthetic: cream/paper background tones, serif headings, monospace accents, terracotta/teal accents (per spec Visual Design).
- Map interaction is country-level only, never city/pin-level (per spec Feature Behavior).
- A country counts as "visited" if and only if it appears in some trip's `countries[]` — never a separately-toggled flag (per spec Data Model rules).
- Creating a trip must copy the shared packing template (`PackingItem` rows with `tripId === null`) into new trip-scoped rows; the template itself must never be mutated by per-trip edits (per spec Data Model rules).
- Storage-layer logic (repository modules, the storage facade, pure utility functions) gets Vitest unit tests; UI/page-level behavior is verified manually by running the dev server (per spec Tech Stack Testing note).
- Currency tips, local-transit tips, and the ranking/badge system are out of scope for this plan — do not add them (per spec Non-goals / Deferred to v2).

```ts
import type { Trip, JournalEntry } from '$lib/types';

const TRIPS_KEY = 'trailune:trips';
const ENTRIES_KEY = 'trailune:journalEntries';

function readAll<T>(key: string): T[] {
  const raw = localStorage.getItem(key);
  return raw ? (JSON.parse(raw) as T[]) : [];
}

function writeAll<T>(key: string, items: T[]): void {
  localStorage.setItem(key, JSON.stringify(items));
}

export function listTrips(): Trip[] {
  return readAll<Trip>(TRIPS_KEY);
}

export function getTrip(id: string): Trip | undefined {
  return listTrips().find((t) => t.id === id);
}

export function saveTrip(trip: Trip): void {
  const trips = listTrips();
  const idx = trips.findIndex((t) => t.id === trip.id);
  if (idx === -1) trips.push(trip);
  else trips[idx] = trip;
  writeAll(TRIPS_KEY, trips);
}

export function deleteTrip(id: string): void {
  writeAll(TRIPS_KEY, listTrips().filter((t) => t.id !== id));
}

export function listJournalEntries(tripId: string): JournalEntry[] {
  return readAll<JournalEntry>(ENTRIES_KEY).filter((e) => e.tripId === tripId);
}

export function getJournalEntry(id: string): JournalEntry | undefined {
  return readAll<JournalEntry>(ENTRIES_KEY).find((e) => e.id === id);
}

export function saveJournalEntry(entry: JournalEntry): void {
  const entries = readAll<JournalEntry>(ENTRIES_KEY);
  const idx = entries.findIndex((e) => e.id === entry.id);
  if (idx === -1) entries.push(entry);
  else entries[idx] = entry;
  writeAll(ENTRIES_KEY, entries);
}

export function deleteJournalEntry(id: string): void {
  writeAll(ENTRIES_KEY, readAll<JournalEntry>(ENTRIES_KEY).filter((e) => e.id !== id));
}

export function deleteJournalEntriesForTrip(tripId: string): void {
  writeAll(ENTRIES_KEY, readAll<JournalEntry>(ENTRIES_KEY).filter((e) => e.tripId !== tripId));
}
```

```ts
import { openDB, type IDBPDatabase } from 'idb';
import type { Photo } from '$lib/types';

let dbPromise: Promise<IDBPDatabase> | undefined;

function getDb(): Promise<IDBPDatabase> {
  if (!dbPromise) {
    dbPromise = openDB('trailune-photos', 1, {
      upgrade(db) {
        db.createObjectStore('photos', { keyPath: 'id' });
      }
    });
  }
  return dbPromise;
}

export async function savePhoto(photo: Photo): Promise<void> {
  const db = await getDb();
  await db.put('photos', photo);
}

export async function getPhoto(id: string): Promise<Photo | undefined> {
  const db = await getDb();
  return db.get('photos', id);
}

export async function deletePhoto(id: string): Promise<void> {
  const db = await getDb();
  await db.delete('photos', id);
}
```

```ts
export function computeThumbnailDimensions(
  width: number,
  height: number,
  maxDim = 400
): { width: number; height: number } {
  if (width <= maxDim && height <= maxDim) return { width, height };
  const scale = maxDim / Math.max(width, height);
  return { width: Math.round(width * scale), height: Math.round(height * scale) };
}

export async function createThumbnail(file: File, maxDim = 400): Promise<Blob> {
  const bitmap = await createImageBitmap(file);
  const { width, height } = computeThumbnailDimensions(bitmap.width, bitmap.height, maxDim);
  const canvas = document.createElement('canvas');
  canvas.width = width;
  canvas.height = height;
  const ctx = canvas.getContext('2d');
  if (!ctx) throw new Error('Canvas 2D context unavailable');
  ctx.drawImage(bitmap, 0, 0, width, height);
  return new Promise((resolve, reject) => {
    canvas.toBlob((blob) => (blob ? resolve(blob) : reject(new Error('Thumbnail encoding failed'))), 'image/jpeg', 0.8);
  });
}
```

```ts
export interface CountryInfo {
  code: string;
  name: string;
}

export function parseCountriesFromSvg(svgText: string): CountryInfo[] {
  const doc = new DOMParser().parseFromString(svgText, 'image/svg+xml');
  const paths = Array.from(doc.querySelectorAll('path[id]'));
  return paths
    .map((path) => {
      const code = path.getAttribute('id') ?? '';
      const title = path.querySelector('title')?.textContent?.trim();
      return { code, name: title && title.length > 0 ? title : code };
    })
    .filter((c) => /^[A-Za-z]{2}$/.test(c.code))
    .sort((a, b) => a.name.localeCompare(b.name));
}

export async function loadWorldCountries(): Promise<CountryInfo[]> {
  const res = await fetch('/world-map.svg');
  const svgText = await res.text();
  return parseCountriesFromSvg(svgText);
}
```

```svelte
<script lang="ts">
  import { page } from '$app/state';
  import { onMount } from 'svelte';
  import { getTrip, listJournalEntries, addJournalEntry } from '$lib/storage/storage';
  import { getPhoto } from '$lib/storage/photos';
  import type { Trip, JournalEntry } from '$lib/types';
  import PackingChecklist from '$lib/components/PackingChecklist.svelte';
  import DoDontList from '$lib/components/DoDontList.svelte';

  let tripId = $derived(page.params.id);
  let trip: Trip | undefined = $state();
  let entries: JournalEntry[] = $state([]);
  let thumbnails: Record<string, string> = $state({});
  let activeTab: 'journal' | 'packing' | 'dodont' = $state('journal');

  let note = $state('');
  let date = $state('');
  let fileInput: HTMLInputElement | undefined = $state();
  let formError: string | null = $state(null);

  const MAX_PHOTO_BYTES = 8 * 1024 * 1024; // 8MB per photo, generous for phone camera JPEGs while keeping localStorage/IndexedDB usage sane

  async function loadEntries() {
    entries = listJournalEntries(tripId).sort((a, b) => b.date.localeCompare(a.date));
    const nextThumbnails = { ...thumbnails };
    for (const entry of entries) {
      for (const photoId of entry.photoIds) {
        if (nextThumbnails[photoId]) continue;
        const photo = await getPhoto(photoId);
        if (photo) nextThumbnails[photoId] = URL.createObjectURL(photo.thumbnailBlob);
      }
    }
    thumbnails = nextThumbnails;
  }

  onMount(async () => {
    trip = getTrip(tripId);
    await loadEntries();
  });

  async function submitEntry(event: SubmitEvent) {
    event.preventDefault();
    formError = null;
    const files = fileInput?.files ? Array.from(fileInput.files) : [];
    const oversized = files.find((f) => f.size > MAX_PHOTO_BYTES);
    if (oversized) {
      formError = `"${oversized.name}" is too large (max 8MB per photo). Please choose a smaller photo.`;
      return;
    }
    try {
      await addJournalEntry(tripId, { date, note, photoFiles: files });
    } catch (err) {
      formError =
        'Could not save this entry with photos — your browser may be blocking local photo storage (common in private browsing). Try again without photos, or in a normal browser window.';
      return;
    }
    note = '';
    date = '';
    if (fileInput) fileInput.value = '';
    await loadEntries();
  }
</script>

{#if trip}

    <p>{trip.startDate} &ndash; {trip.endDate}</p>

      {#each entries as entry (entry.id)}

      </form>
    </section>
```

```svelte
<script lang="ts">
  import { page } from '$app/state';
  import { goto } from '$app/navigation';
  import { onMount } from 'svelte';
  import { getJournalEntry, saveJournalEntry, deleteJournalEntry } from '$lib/storage/storage';
  import { getPhoto } from '$lib/storage/photos';
  import type { JournalEntry } from '$lib/types';

  let tripId = $derived(page.params.id);
  let entryId = $derived(page.params.entryId);
  let entry: JournalEntry | undefined = $state();
  let photoUrls: string[] = $state([]);
  let note = $state('');
  let date = $state('');

  onMount(async () => {
    entry = getJournalEntry(entryId);
    if (entry) {
      note = entry.note;
      date = entry.date;
      const urls: string[] = [];
      for (const photoId of entry.photoIds) {
        const photo = await getPhoto(photoId);
        if (photo) urls.push(URL.createObjectURL(photo.blob));
      }
      photoUrls = urls;
    }
  });

  function save(event: SubmitEvent) {
    event.preventDefault();
    if (!entry) return;
    saveJournalEntry({ ...entry, note, date });
    goto(`/trips/${tripId}`);
  }

  function remove() {
    if (!entry) return;
    deleteJournalEntry(entry.id);
    goto(`/trips/${tripId}`);
  }
</script>

{#if entry}

    {/each}
  </div>
  <form onsubmit={save}>
    <label>Date <input type="date" bind:value={date} required /></label>
    <label>Note <textarea bind:value={note} required></textarea></label>
    <button type="submit">Save</button>
  </form>
  <button onclick={remove}>Delete entry</button>
{:else}
  <p>Entry not found.</p>
{/if}
```
