# Dream Map Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A new `/dream-map` page lets a user click not-yet-visited countries to rate (1-5 stars) and note them as dream destinations, shown on their own map in a distinct color plus a sorted list below it.

**Architecture:** A `WishlistEntry` type and a `localStorage`-backed `wishlist.ts` storage module (mirroring the existing `pins.ts` pattern) hold the data. A new `DreamMap.svelte` component (a lighter sibling of `WorldMap.svelte`, not a mode of it) renders the map with click-to-zoom and three visual states (visited/muted, wishlisted/accent, default/muted-clickable). The `/dream-map` page wires trips + wishlist data together, filtering out any country that has since become visited, and renders the map, an edit panel, and a sorted list.

**Tech Stack:** SvelteKit 2, Svelte 5 runes, TypeScript, Vitest.

## Global Constraints

- No color-by-rating scale — every wishlisted country uses the single `--color-wishlist` CSS var regardless of rating (spec: Non-goals, Goals).
- No hover popover, no flight-route, no reveal-animation on this page (spec: Non-goals).
- A country becomes ineligible for the wishlist display the moment it's in `visitedCodes` — filtered at render time only, never by deleting the underlying `WishlistEntry` (spec: Non-goals, Edge cases). Deleting the trip that made it visited must make the original entry reappear automatically.
- Save is only meaningful once a rating of 1-5 is selected — no zero-star entries are stored (spec: Edge cases).
- Clicking an already-visited country must be a no-op — no click listener attached to `.visited` paths at all (spec: Edge cases).

Create `src/lib/storage/wishlist.ts`:

```ts
import type { WishlistEntry } from '$lib/types';

const WISHLIST_KEY = 'trailune:wishlist';

function readAll(): WishlistEntry[] {
  const raw = localStorage.getItem(WISHLIST_KEY);
  return raw ? (JSON.parse(raw) as WishlistEntry[]) : [];
}

function writeAll(items: WishlistEntry[]): void {
  localStorage.setItem(WISHLIST_KEY, JSON.stringify(items));
}

export function listWishlist(): WishlistEntry[] {
  return readAll();
}

export function saveWishlistEntry(entry: WishlistEntry): void {
  const items = readAll();
  const idx = items.findIndex((e) => e.id === entry.id);
  if (idx === -1) items.push(entry);
  else items[idx] = entry;
  writeAll(items);
}

export function deleteWishlistEntry(id: string): void {
  writeAll(readAll().filter((e) => e.id !== id));
}
```

Create `src/lib/components/DreamMap.svelte`:

```svelte
<script lang="ts">
  import { onMount } from 'svelte';
  import { tweened } from 'svelte/motion';
  import { cubicOut } from 'svelte/easing';
  import Skeleton from './Skeleton.svelte';
  import { padBBox, type ViewBox } from '$lib/mapZoom';

  let {
    visitedCodes,
    wishlistCodes,
    oncountryclick
  }: {
    visitedCodes: Set<string>;
    wishlistCodes: Set<string>;
    oncountryclick?: (code: string) => void;
  } = $props();

  let svgContainer: HTMLDivElement | undefined = $state();
  let loaded = $state(false);
  let svgEl: SVGSVGElement | undefined;
  let originalViewBox: ViewBox = [0, 0, 2000, 900];

  const viewBoxTween = tweened<ViewBox>([0, 0, 2000, 900], { duration: 500, easing: cubicOut });

  function prefersReducedMotion(): boolean {
    return window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  }

  function syncClasses() {
    if (!svgContainer) return;
    svgContainer.querySelectorAll('path[id]').forEach((path) => {
      const code = path.id;
      path.classList.toggle('visited', visitedCodes.has(code));
      path.classList.toggle('wishlisted', wishlistCodes.has(code));
    });
  }

  onMount(async () => {
    const res = await fetch('/world-map.svg');
    const svgText = await res.text();
    if (!svgContainer) return;
    svgContainer.innerHTML = svgText;
    const svg = svgContainer.querySelector('svg');
    svg?.setAttribute('class', 'world-map-svg');
    if (svg) {
      svgEl = svg as unknown as SVGSVGElement;
      const vb = svg.getAttribute('viewBox');
      if (vb) {
        const parts = vb.split(/\s+/).map(Number) as ViewBox;
        originalViewBox = parts;
        viewBoxTween.set(parts, { duration: 0 });
      }
    }
    loaded = true;
    syncClasses();
    svgContainer.querySelectorAll('path[id]').forEach((path) => {
      const code = path.id;
      if (visitedCodes.has(code)) return;
      path.addEventListener('click', () => {
        oncountryclick?.(code);
        const bbox = (path as unknown as SVGGraphicsElement).getBBox();
        viewBoxTween.set(padBBox(bbox), prefersReducedMotion() ? { duration: 0 } : undefined);
      });
    });
  });

  $effect(() => {
    syncClasses();
  });

  $effect(() => {
    if (svgEl) {
      svgEl.setAttribute('viewBox', $viewBoxTween.join(' '));
    }
  });

  function handleOutsideClick(event: PointerEvent) {
    const target = event.target as Element;
    if (target.closest('path[id]')) return;
    viewBoxTween.set(originalViewBox, prefersReducedMotion() ? { duration: 0 } : undefined);
  }
</script>

<svelte:window onpointerdown={handleOutsideClick} />

{#if !loaded}
  <div style="aspect-ratio: 2000 / 900;">
    <Skeleton width="100%" height="100%" radius="var(--radius-xl)" />
  </div>
{/if}
<div class="world-map" bind:this={svgContainer} class:hidden={!loaded}></div>

<style>
  .world-map.hidden {
    display: none;
  }

  .world-map :global(.world-map-svg) {
    width: 100%;
    height: auto;
  }
  .world-map :global(path) {
    fill: var(--color-ink-band-soft);
    stroke: var(--color-ink-band);
    stroke-width: 0.4;
    transition:
      fill 0.2s var(--ease-out-quart),
      filter 0.2s var(--ease-out-quart);
  }
  .world-map :global(path.wishlisted) {
    fill: var(--color-wishlist);
  }
  .world-map :global(path:not(.visited)) {
    cursor: pointer;
  }
  .world-map :global(path:not(.visited):hover) {
    filter: brightness(1.15);
  }

  @media (prefers-reduced-motion: reduce) {
    .world-map :global(path) {
      transition: fill 0.2s ease;
    }
    .world-map :global(path:not(.visited):hover) {
      filter: none;
    }
  }
</style>
```

Create `src/routes/dream-map/+page.svelte`:

```svelte
<script lang="ts">
  import { onMount } from 'svelte';
  import DreamMap from '$lib/components/DreamMap.svelte';
  import StarRating from '$lib/components/StarRating.svelte';
  import { listTrips } from '$lib/storage/storage';
  import { loadWorldCountries, type CountryInfo } from '$lib/countries';
  import { listWishlist, saveWishlistEntry, deleteWishlistEntry } from '$lib/storage/wishlist';
  import { generateId } from '$lib/utils/id';
  import type { Trip, WishlistEntry } from '$lib/types';

  let trips: Trip[] = $state([]);
  let countries: CountryInfo[] = $state([]);
  let entries: WishlistEntry[] = $state([]);
  let selectedCode: string | null = $state(null);
  let panelRating: number | undefined = $state();
  let panelNote = $state('');

  let visitedCodes = $derived(new Set(trips.flatMap((t) => t.countries)));
  let visibleEntries = $derived(entries.filter((e) => !visitedCodes.has(e.countryCode)));
  let wishlistCodes = $derived(new Set(visibleEntries.map((e) => e.countryCode)));
  let sortedEntries = $derived([...visibleEntries].sort((a, b) => b.rating - a.rating));
  let selectedEntry = $derived(
    selectedCode ? (visibleEntries.find((e) => e.countryCode === selectedCode) ?? null) : null
  );
  let selectedName = $derived(
    selectedCode ? (countries.find((c) => c.code === selectedCode)?.name ?? selectedCode) : ''
  );

  function countryName(code: string): string {
    return countries.find((c) => c.code === code)?.name ?? code;
  }

  onMount(async () => {
    trips = listTrips();
    countries = await loadWorldCountries();
    entries = listWishlist();
  });

  function openPanel(code: string) {
    selectedCode = code;
    const existing = visibleEntries.find((e) => e.countryCode === code);
    panelRating = existing?.rating;
    panelNote = existing?.note ?? '';
  }

  function closePanel() {
    selectedCode = null;
    panelRating = undefined;
    panelNote = '';
  }

  function handleSave() {
    if (!selectedCode || panelRating === undefined) return;
    const existing = visibleEntries.find((e) => e.countryCode === selectedCode);
    const entry: WishlistEntry = {
      id: existing?.id ?? generateId(),
      countryCode: selectedCode,
      rating: panelRating,
      note: panelNote,
      createdAt: existing?.createdAt ?? new Date().toISOString()
    };
    saveWishlistEntry(entry);
    entries = listWishlist();
    closePanel();
  }

  function handleDelete() {
    if (!selectedEntry) return;
    deleteWishlistEntry(selectedEntry.id);
    entries = listWishlist();
    closePanel();
  }
</script>

<div class="mx-auto max-w-3xl px-4 py-8">
  <h1 class="font-display text-3xl font-bold sm:text-4xl">Dream Map</h1>
  <p class="mt-1 text-[var(--color-ink-soft)]">
    Countries you haven't visited yet, rated and noted for someday.
  </p>

  <div class="mt-6 rounded-[var(--radius-xl)] bg-[var(--color-ink-band)] p-4">
    <DreamMap {visitedCodes} {wishlistCodes} oncountryclick={openPanel} />
  </div>

  {#if selectedCode}
    <div class="mt-4 rounded-[var(--radius-lg)] border border-[var(--color-card-border)] bg-[var(--color-card)] p-4">
      <div class="flex items-center justify-between">
        <h2 class="font-semibold text-[var(--color-ink)]">{selectedName}</h2>
        <button type="button" class="text-sm text-[var(--color-ink-soft)]" onclick={closePanel}>
          Close
        </button>
      </div>
      <div class="mt-3">
        <StarRating bind:value={panelRating} />
      </div>
      <textarea
        bind:value={panelNote}
        placeholder="Why is this a dream destination?"
        rows="2"
        class="mt-3 w-full rounded-[var(--radius-md)] border border-[var(--color-card-border)] bg-[var(--color-canvas)] p-2 text-sm text-[var(--color-ink)]"
      ></textarea>
      <div class="mt-3 flex gap-2">
        <button
          type="button"
          class="rounded-[var(--radius-md)] bg-[var(--color-accent)] px-3 py-1.5 text-sm font-medium text-[var(--color-accent-ink)] disabled:opacity-50"
          disabled={panelRating === undefined}
          onclick={handleSave}
        >
          Save
        </button>
        {#if selectedEntry}
          <button
            type="button"
            class="rounded-[var(--radius-md)] border border-[var(--color-danger)] px-3 py-1.5 text-sm font-medium text-[var(--color-danger)]"
            onclick={handleDelete}
          >
            Delete
          </button>
        {/if}
      </div>
    </div>
  {/if}

  <div class="mt-8">
    <h2 class="font-semibold text-[var(--color-ink)]">Your dream destinations</h2>
    {#if sortedEntries.length === 0}
      <p class="mt-2 text-sm text-[var(--color-ink-soft)]">
        No dream destinations yet — click a country on the map to add one.
      </p>
    {:else}
      <ul class="mt-2 space-y-2">
        {#each sortedEntries as entry (entry.id)}
          <li>
            <button
              type="button"
              class="flex w-full items-center justify-between gap-2 rounded-[var(--radius-md)] border border-[var(--color-card-border)] bg-[var(--color-card)] px-3 py-2 text-left"
              onclick={() => openPanel(entry.countryCode)}
            >
              <span class="font-medium text-[var(--color-ink)]">{countryName(entry.countryCode)}</span>
              <StarRating value={entry.rating} readonly />
            </button>
            {#if entry.note}
              <p class="mt-1 px-1 text-sm text-[var(--color-ink-soft)]">{entry.note}</p>
            {/if}
          </li>
        {/each}
      </ul>
    {/if}
  </div>
</div>
```
