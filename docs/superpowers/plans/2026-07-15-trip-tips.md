# Trip Tips (Local Tips & Sightseeing) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a "Tips" tab to the trip hub where the user can log free-text local tips (currency, places to avoid, getting around, sightseeing), each optionally attached to a single photo.

**Architecture:** A new `TipEntry` type and a `tips.ts` storage repository (mirroring the existing `local.ts`/`checklist.ts` repositories) hold the data. The `storage.ts` facade adds `addTipEntry`/`deleteTipEntryWithPhoto` and extends `deleteTripCascade`. A new `TipsList.svelte` component (mirroring `DoDontList.svelte`'s self-contained pattern) renders the tab, wired into the trip hub's existing tab bar.

**Tech Stack:** SvelteKit (Svelte 5 runes), TypeScript, Tailwind CSS v4, Vitest, `idb` (existing photo store).

## Global Constraints

- No backend/API calls — all client-side (localStorage for `TipEntry`, existing IndexedDB photo store for the optional photo), per `docs/superpowers/specs/2026-07-15-trip-tips-design.md`.
- Tips are scoped to the whole trip, not per-country.
- One optional photo per tip entry (not an array like Journal).
- UI changes (the new tab, the `TipsList` component) are verified manually via the dev server — this project has no component-level test suite; only the storage layer has Vitest coverage.

```ts
import type { TipEntry } from '$lib/types';

const TIPS_KEY = 'trailune:tips';

function readAll(): TipEntry[] {
  const raw = localStorage.getItem(TIPS_KEY);
  return raw ? (JSON.parse(raw) as TipEntry[]) : [];
}

function writeAll(tips: TipEntry[]): void {
  localStorage.setItem(TIPS_KEY, JSON.stringify(tips));
}

export function listTips(tripId: string): TipEntry[] {
  return readAll().filter((t) => t.tripId === tripId);
}

export function getTip(id: string): TipEntry | undefined {
  return readAll().find((t) => t.id === id);
}

export function saveTip(tip: TipEntry): void {
  const tips = readAll();
  const idx = tips.findIndex((t) => t.id === tip.id);
  if (idx === -1) tips.push(tip);
  else tips[idx] = tip;
  writeAll(tips);
}

export function deleteTip(id: string): void {
  writeAll(readAll().filter((t) => t.id !== id));
}

export function deleteTipsForTrip(tripId: string): void {
  writeAll(readAll().filter((t) => t.tripId !== tripId));
}
```

```ts
export async function addTipEntry(
  tripId: string,
  input: { category: TipCategory; note: string; photoFile: File | null }
): Promise<TipEntry> {
  let photoId: string | null = null;
  if (input.photoFile) {
    photoId = generateId();
    let thumbnailBlob: Blob;
    try {
      thumbnailBlob = await createThumbnail(input.photoFile);
    } catch {
      thumbnailBlob = input.photoFile;
    }
    await photoStore.savePhoto({
      id: photoId,
      blob: input.photoFile,
      thumbnailBlob,
      createdAt: new Date().toISOString()
    });
  }

  const tip: TipEntry = {
    id: generateId(),
    tripId,
    category: input.category,
    note: input.note,
    photoId,
    createdAt: new Date().toISOString()
  };
  tipsRepo.saveTip(tip);
  return tip;
}

export async function deleteTipEntryWithPhoto(id: string): Promise<void> {
  const tip = tipsRepo.getTip(id);
  if (tip?.photoId) {
    await photoStore.deletePhoto(tip.photoId);
  }
  tipsRepo.deleteTip(id);
}
```

Replace the existing `deleteTripCascade` function with this version (adds tip cleanup before the final `trips.deleteTrip(tripId)` call; everything else is unchanged):

```ts
export async function deleteTripCascade(tripId: string): Promise<void> {
  const entries = trips.listJournalEntries(tripId);
  for (const entry of entries) {
    for (const photoId of entry.photoIds) {
      await photoStore.deletePhoto(photoId);
    }
  }
  trips.deleteJournalEntriesForTrip(tripId);
  checklist.deletePackingItemsForTrip(tripId);
  checklist.deleteDoDontItemsForTrip(tripId);

  const tripTips = tipsRepo.listTips(tripId);
  for (const tip of tripTips) {
    if (tip.photoId) {
      await photoStore.deletePhoto(tip.photoId);
    }
  }
  tipsRepo.deleteTipsForTrip(tripId);

  trips.deleteTrip(tripId);
}
```

```svelte
<script lang="ts">
  import { onMount } from 'svelte';
  import { listTips, addTipEntry, deleteTipEntryWithPhoto } from '$lib/storage/storage';
  import { getPhoto } from '$lib/storage/photos';
  import type { TipEntry, TipCategory } from '$lib/types';

  let { tripId }: { tripId: string } = $props();

  const MAX_PHOTO_BYTES = 8 * 1024 * 1024;

  const categories: Array<{ value: TipCategory; label: string }> = [
    { value: 'currency', label: 'Currency' },
    { value: 'avoid', label: 'Avoid' },
    { value: 'transit', label: 'Getting Around' },
    { value: 'sightseeing', label: 'Sightseeing' }
  ];

  let tips: TipEntry[] = $state([]);
  let thumbnails: Record<string, string> = $state({});
  let category: TipCategory = $state('currency');
  let note = $state('');
  let fileInput: HTMLInputElement | undefined = $state();
  let formError: string | null = $state(null);
  let submitting = $state(false);

  async function load() {
    tips = listTips(tripId);
    const nextThumbnails = { ...thumbnails };
    for (const tip of tips) {
      if (!tip.photoId || nextThumbnails[tip.photoId]) continue;
      const photo = await getPhoto(tip.photoId);
      if (photo) nextThumbnails[tip.photoId] = URL.createObjectURL(photo.thumbnailBlob);
    }
    thumbnails = nextThumbnails;
  }

  onMount(load);

  function tipsFor(value: TipCategory): TipEntry[] {
    return tips.filter((t) => t.category === value);
  }

  async function submit(event: SubmitEvent) {
    event.preventDefault();
    formError = null;
    const file = fileInput?.files?.[0] ?? null;
    if (file && file.size > MAX_PHOTO_BYTES) {
      formError = `"${file.name}" is too large (max 8MB per photo). Please choose a smaller photo.`;
      return;
    }
    submitting = true;
    try {
      await addTipEntry(tripId, { category, note, photoFile: file });
    } catch {
      formError =
        'Could not save this tip with a photo — your browser may be blocking local photo storage (common in private browsing). Try again without a photo, or in a normal browser window.';
      submitting = false;
      return;
    }
    note = '';
    if (fileInput) fileInput.value = '';
    await load();
    submitting = false;
  }

  async function remove(id: string) {
    await deleteTipEntryWithPhoto(id);
    await load();
  }
</script>

          bind:value={category}
          class="rounded-[var(--radius-md)] border border-[var(--color-card-border)] bg-[var(--color-canvas)] px-3.5 py-2.5 text-[var(--color-ink)] outline-none transition-shadow focus:border-[var(--color-accent)] focus:ring-2 focus:ring-[var(--color-accent-soft)]"

          {/each}

      </label>
```
